class Solution {
public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) {
        vector<vector<int>> dis(n, vector<int>(n, INT_MAX));
        for(auto i: edges)
        {
            int a = i[0], b = i[1], weight = i[2];
            dis[a][b] = weight;
            dis[b][a] = weight;
        }
        for(int i = 0; i < n; i++)
            dis[i][i] = 0;
        for(int k = 0; k < n; k++)
        for(int i = 0; i < n; i++)
        for(int j = 0; j < n; j++)
        {
            if(dis[i][k] == INT_MAX or dis[k][j] == INT_MAX)
                continue;
            dis[i][j] = min(dis[i][k] + dis[k][j], dis[i][j]);
        }
        int cityno = n, city;
        for(int i = 0; i < n; i++)
        {
            int cnt = 0; 
            for(int j = 0; j < n; j++)
            {
                if(dis[i][j] <= distanceThreshold)
                    cnt++;
            }
            if(cnt <= cityno)
            {
                cityno = cnt;
                city = i;
            }
        }
        return city;
    }
};
